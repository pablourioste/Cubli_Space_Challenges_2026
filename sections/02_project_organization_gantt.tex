% Cubli - consolidated project schedule.
%
% Day numbering: day 1 = 23 Jul 2026, day 29 = 20 Aug 2026.
%
% The "today" marker is computed from the compilation date rather than being
% written in by hand, so the chart tracks the real calendar on every build.
% \cubliday holds the project day number for today, clamped to the 1-29 chart
% range; outside the project window the marker is simply not drawn.
% ---------------------------------------------------------

% Project day number for the compilation date.
% Day 1 is 23 July 2026, whose Julian day number is 2461247; subtracting
% 2461246 therefore maps that date to day 1.
% Converts \year/\month/\day to a Julian day number (Fliegel-Van Flandern) and
% offsets it so that 23 July 2026 is day 1. Every \advance, \multiply and
% \divide is terminated with \relax so that TeX stops scanning the number.
\begingroup
\count0=\year \count2=\month \count4=\day
\count6=\count2 \advance\count6 by -14\relax \divide\count6 by 12\relax
\count8=\count0 \advance\count8 by 4800\relax \advance\count8 by \count6
\count10=\count2 \advance\count10 by -2\relax
\count11=\count6 \multiply\count11 by -12\relax \advance\count10 by \count11
\count12=\count8 \multiply\count12 by 1461\relax \divide\count12 by 4\relax
\count14=\count10 \multiply\count14 by 367\relax \divide\count14 by 12\relax
\count16=\count8 \advance\count16 by 100\relax
\divide\count16 by 100\relax
\multiply\count16 by 3\relax
\divide\count16 by 4\relax
\count18=\count12 \advance\count18 by \count14 \advance\count18 by -\count16
\advance\count18 by \count4 \advance\count18 by -32075\relax
\advance\count18 by -2461244\relax   % maps 23 Jul 2026 to day 1
\xdef\cubliday{\the\count18}
\endgroup

% The today marker is set from \cubliday, computed above, and is applied only
% when the compilation date falls inside the project window. Outside that
% window no marker is drawn and the chart is otherwise unchanged.
\ifnum\cubliday>0
  \ifnum\cubliday<30
    \ganttset{today=\cubliday}%
  \fi
\fi
