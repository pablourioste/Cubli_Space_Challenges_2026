% ---------- Figure placeholders (pending artwork) ----------
% A framed, hatched box standing in for artwork that is not yet drawn, so that
% the figure is numbered, captioned and listed in the List of Figures before the
% file exists. Replacing a placeholder is a one-line edit: swap \phfig for
% \includegraphics and keep the surrounding float, caption and label.
%   \phfig[<height>]{<width>}{<what the drawing will show>}
\usepackage{tikz}
\usetikzlibrary{patterns}
\newcommand{\phfig}[3][0.42\textheight]{%
  \begin{tikzpicture}
    \node[draw=black!45, dashed, line width=0.8pt, fill=black!3,
          minimum width=#2, minimum height=#1, inner sep=6pt,
          align=center, text width=0.86\dimexpr#2\relax, font=\small]
         {\textcolor{red}{\textbf{PLACEHOLDER --- artwork pending}}\\[6pt]#3};
  \end{tikzpicture}%
}

% ---------- Header / Footer ----------
% Team mark for the running header. The source PNG is the full cover logo: a
% square canvas with the cube mark in the upper half, the "ARMONIA" wordmark
% below it, and wide white padding on all sides. At header size the wordmark
% would be a smudge, so the image is trimmed to the cube mark alone. The trim
% is given in fractions of the natural image size (adjustbox relative units, in
% the order left/bottom/right/top), so re-exporting the logo at a different
% resolution does not break the crop -- only a change to the artwork's layout
% would. To show the whole logo instead, drop the trim/clip keys and raise
% \headlogoheight. \adjincludegraphics rather than \includegraphics: the
% relative \width/\height trim units are an adjustbox feature.
\newlength{\headlogoheight}
\setlength{\headlogoheight}{28pt}
\newcommand{\headlogo}{%
    % The declared depth (10pt) exceeds the visual one (the 6pt lower), because
    % fancyhdr draws the header rule flush under the deepest header material:
    % the extra 4pt is what keeps the logo from sitting on the rule.
    \raisebox{-6pt}[24pt][10pt]{%
        \adjincludegraphics[trim={0.29\width} {0.46\height} {0.30\width} {0.155\height}, clip, height=\headlogoheight]{images/armonia_logo.png}%
    }%
}
